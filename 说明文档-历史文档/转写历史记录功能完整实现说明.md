```python
@Slot(str, result=str)
def saveTranscriptSession(self, session_data_json):
    """保存转写会话到文件"""
    # 解析会话数据
    session_data = json.loads(session_data_json)
    
    # 创建audio_text文件夹
    audio_text_dir = os.path.join(os.getcwd(), 'audio_text')
    os.makedirs(audio_text_dir, exist_ok=True)
    
    # 生成文件名：transcript_YYYYMMDD_HHMMSS.json
    now = datetime.now()
    filename = f"transcript_{now.strftime('%Y%m%d_%H%M%S')}.json"
    file_path = os.path.join(audio_text_dir, filename)
    
    # 保存到文件
    with open(file_path, 'w', encoding='utf-8') as f:
        json.dump(session_data, f, ensure_ascii=False, indent=2)
    
    return file_path
```

#### **loadTranscriptHistory()**
```python
@Slot(result=str)
def loadTranscriptHistory(self):
    """载入历史转写记录"""
    from PySide6.QtWidgets import QFileDialog
    
    # 创建audio_text文件夹（如果不存在）
    audio_text_dir = os.path.join(os.getcwd(), 'audio_text')
    os.makedirs(audio_text_dir, exist_ok=True)
    
    # 打开文件选择对话框
    file_path, _ = QFileDialog.getOpenFileName(
        self.main_window,
        "选择转写历史记录",
        audio_text_dir,
        "JSON文件 (*.json);;所有文件 (*.*)"
    )
    
    if not file_path:
        return ""
    
    # 读取文件内容
    with open(file_path, 'r', encoding='utf-8') as f:
        session_data = json.load(f)
    
    return json.dumps(session_data, ensure_ascii=False)
```

## 🔧 **技术实现细节**

### **前端核心函数**

#### **会话管理**
```javascript
// 开始新会话
function startNewTranscriptSession() {
    const now = new Date();
    currentTranscriptSession = {
        sessionId: 'transcript_' + now.getTime(),
        startTime: now.toISOString(),
        transcripts: [],
        filePath: null
    };
}

// 实时保存
function saveTranscriptSession() {
    if (window.bridge && window.bridge.saveTranscriptSession) {
        const sessionData = {
            sessionId: currentTranscriptSession.sessionId,
            startTime: currentTranscriptSession.startTime,
            transcripts: currentTranscriptSession.transcripts,
            saveTime: new Date().toISOString()
        };
        
        window.bridge.saveTranscriptSession(JSON.stringify(sessionData), function(filePath) {
            if (filePath) {
                currentTranscriptSession.filePath = filePath;
                debugLog('转写会话已保存到: ' + filePath);
            }
        });
    }
}
```

#### **历史载入**
```javascript
// 载入历史记录
window.loadTranscriptHistory = function() {
    if (window.bridge && window.bridge.loadTranscriptHistory) {
        window.bridge.loadTranscriptHistory(function(result) {
            if (result) {
                try {
                    const sessionData = JSON.parse(result);
                    loadTranscriptSessionData(sessionData);
                    debugLog('历史记录载入成功');
                } catch (error) {
                    alert('载入历史记录失败：文件格式错误');
                }
            }
        });
    }
};

// 重建界面
function loadTranscriptSessionData(sessionData) {
    // 清空当前内容
    const transcriptDiv = document.getElementById('transcript-content');
    transcriptDiv.innerHTML = '';
    
    // 更新会话信息
    currentTranscriptSession = {
        sessionId: sessionData.sessionId,
        startTime: sessionData.startTime,
        transcripts: sessionData.transcripts || [],
        filePath: null
    };
    
    // 重建所有转写条目
    sessionData.transcripts.forEach(record => {
        addTranscriptItemToUI(record.timestamp, record.text, record.summarized);
    });
}
```

### **状态保持机制**

#### **总结状态标记**
```javascript
function markTranscriptItemsAsSummarized() {
    selectedTranscriptTexts.forEach(item => {
        // 添加UI标记
        const timestamp = item.element.querySelector('span');
        if (!timestamp.querySelector('.summarized-mark')) {
            const checkMark = document.createElement('span');
            checkMark.className = 'summarized-mark text-green-500 ml-1';
            checkMark.innerHTML = '✓';
            checkMark.title = '已总结';
            timestamp.appendChild(checkMark);
        }
        
        // 更新会话数据
        const transcriptRecord = currentTranscriptSession.transcripts.find(record => 
            record.text === item.text && record.timestamp === item.timestamp
        );
        if (transcriptRecord) {
            transcriptRecord.summarized = true;
        }
    });
    
    // 保存更新
    saveTranscriptSession();
}
```
