## 技术实现 🔧

### 1. 颜色映射函数
```javascript
function getKnowledgePointColor(masteryScore) {
    // 未评估处理
    if (masteryScore === -1 || masteryScore === null || masteryScore === undefined) {
        return '#E5E7EB'; // 浅灰色
    }
    
    // 分数范围限制
    const score = Math.max(0, Math.min(100, masteryScore));
    
    // 颜色数组：从极浅橙色到深酒红色
    const colors = [
        '#FFF7ED', // 0-10分：极浅橙色
        '#FFEDD5', // 10-20分：淡桃色
        '#FED7AA', // 20-30分：浅杏色
        '#FDBA74', // 30-40分：明显橙色
        '#FB923C', // 40-50分：鲜明橙色
        '#F97316', // 50-60分：浓郁橙色
        '#EA580C', // 60-70分：深红橙
        '#DC2626', // 70-80分：橙红色
        '#B91C1C', // 80-90分：深红色
        '#991B1B'  // 90-100分：深酒红
    ];
    
    // 计算并返回颜色
    const colorIndex = Math.floor(score / 10);
    return colors[Math.min(colorIndex, colors.length - 1)];
}
```

### 2. 智能文字颜色函数
```javascript
function getTextColor(masteryScore) {
    // 40分以下：浅色背景，使用深色文字
    if (masteryScore === -1 || masteryScore === null || 
        masteryScore === undefined || masteryScore < 40) {
        return '#374151'; // 深灰色文字
    } else {
        // 40分以上：深色背景，使用白色文字
        return '#FFFFFF'; // 白色文字
    }
}
```

### 3. D3.js渲染中的应用
```javascript
// 节点颜色设置
.style('fill', d => {
    if (d.type === 'knowledge_point') {
        return getKnowledgePointColor(d.mastery_score);
    }
    return '#F59E0B'; // 默认颜色
})

// 文字颜色设置
.style('fill', d => {
    if (d.type === 'knowledge_point') {
        const masteryScore = d.mastery_score;
        return masteryScore < 40 ? '#374151' : '#FFFFFF';
    }
    return 'white';
})
```
